Run("partial",(r,s,c,b)=>{var e=c.Session.SessionId;var z=c.ApplyComposition([CompositionEvent.Result(e,0,0,0,"X",new(0,1)),CompositionEvent.Start(e,1,999,0,default,"bad")]);Check(b.Text=="Xb"&&z.Handled&&z.Outcome.HasFlag(TextInputOutcome.ResultCommitted)&&z.Outcome.HasFlag(TextInputOutcome.Rejected),"partial batch reports actual write");});

Run("getterThrow",(r,s,c,b)=>{var ev=CompositionEvent.Result(c.Session.SessionId,0,0,0,"X",new(0,1));b.ThrowGet=true;var z=c.ApplyComposition([ev]);b.ThrowGet=false;Check(z.HasRejection&&b.Writes==0,"getter exception contained in callback");});
