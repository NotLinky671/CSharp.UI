using CSharp.UI;
using CSharp.UI.Testing;

int passed=0,failed=0;
void Check(bool ok,string name){Console.WriteLine($"{(ok?"PASS":"FAIL")} {name}");if(ok)passed++;else failed++;}
void Run(string name,Action<Field> test,string initial="",int max=int.MaxValue,bool input=true){using var f=new Field(initial,max,input);try{test(f);}catch(Exception e){f.Box.ThrowGet=false;f.Box.ThrowAfterWrite=false;Check(false,name+" threw "+e.GetType().Name);}}
foreach(var ms in new[]{499,500,501})Run("merge"+ms,f=>{f.Text("a");f.Clock.Advance(TimeSpan.FromMilliseconds(ms));f.Text("b");f.Undo();Check(f.Value==(ms<=500?"":"a"),$"typing merge {ms}ms");f.Redo();Check(f.Value=="ab",$"typing redo {ms}ms");});
Run("selection",f=>{f.Key(UiKey.Left,KeyModifiers.Shift);f.Key(UiKey.Left,KeyModifiers.Shift);f.Text("X");Check(f.Value=="abX","selection replacement");f.Undo();Check(f.Value=="abcd","undo selection text");var s=f.Client.Document.Selection;Check(s.Anchor==4&&s.Active==2,"undo backward selection direction");f.Redo();Check(f.Value=="abX"&&f.Client.Document.Selection==TextSelection.Caret(3),"redo after selection");},"abcd");
Run("moveBack",f=>{f.Text("a");f.Key(UiKey.Left);f.Key(UiKey.Right);f.Text("b");f.Undo();Check(f.Value=="a","moving away and back breaks merge");});
Run("copy",f=>{f.Text("a");f.Key(UiKey.C,Field.Command);f.Text("b");f.Undo();Check(f.Value=="a","copy command breaks merge even without selection");});
Run("redo",f=>{f.Text("a");f.Undo();f.Text("\n\t");f.Redo();Check(f.Value=="a","scrubbed noop preserves redo");f.Undo();f.Text("b");f.Redo();Check(f.Value=="b","accepted new edit clears redo");});
Run("external",f=>{f.Text("a");f.Box.Value="external";var writes=f.Box.Writes;f.Undo();Check(f.Value=="external"&&f.Box.Writes==writes,"external binding update clears old undo before write");});
Run("capacity",f=>{f.Text("bc");Check(f.Value=="ab","MaxLength accepts whole available fragment");f.Undo();Check(f.Value=="a","undo uses actual accepted fragment");},"a",2);
Run("rejectedCapacity",f=>{f.Text("a");f.Undo();f.Clipboard.SetText("😀");f.Key(UiKey.V,Field.Command);f.Redo();Check(f.Value=="a","capacity-rejected paste preserves redo");},"",1);
Run("reject",f=>{f.Text("a");f.Box.Reject=true;var count=f.Box.Writes;f.Undo();Check(f.Value=="a"&&f.Box.Writes==count+1,"undo setter rejection is actual unchanged value");f.Box.Reject=false;count=f.Box.Writes;f.Undo();f.Redo();Check(f.Value=="a"&&f.Box.Writes==count,"rejected undo clears untrusted stacks");});
Run("normalize",f=>{f.Text("a");f.Clock.Advance(TimeSpan.FromSeconds(1));f.Text("b");f.Box.Transform=v=>v.ToUpperInvariant();f.Undo();Check(f.Value=="A","normalized undo reads actual binding");f.Box.Transform=null;var n=f.Box.Writes;f.Undo();f.Redo();Check(f.Value=="A"&&f.Box.Writes==n,"normalized undo clears both stacks");});
Run("throw",f=>{f.Text("a");f.Box.ThrowAfterWrite=true;try{f.Undo();}catch(InvalidOperationException){}f.Box.ThrowAfterWrite=false;var n=f.Box.Writes;f.Redo();f.Undo();Check(f.Value==""&&f.Box.Writes==n,"write-then-throw leaves actual document and clears history");});
Run("getter",f=>{f.Text("a");f.Box.ThrowGet=true;try{f.Undo();}catch(InvalidOperationException){}f.Box.ThrowGet=false;var n=f.Box.Writes;f.Undo();f.Redo();Check(f.Value=="a"&&f.Box.Writes==n,"getter failure clears untrusted history");});
Run("clientGetter",f=>{f.Text("a");f.Box.ThrowGet=true;_ = f.Client.Document;f.Box.ThrowGet=false;var n=f.Box.Writes;f.Undo();f.Redo();Check(f.Value=="a"&&f.Box.Writes==n,"composition client read failure also invalidates history");});
Run("grapheme",f=>{f.Text("e");f.Clock.Advance(TimeSpan.FromSeconds(1));f.Text("\u0301");f.Undo();Check(f.Value=="e","undo cross-transaction combining mark");f.Undo();Check(f.Value=="","undo original base letter");f.Redo();f.Redo();Check(f.Value=="e\u0301","redo combining mark restores complete text");});
Run("ZWJ",f=>{f.Text("👩");f.Clock.Advance(TimeSpan.FromSeconds(1));f.Text("\u200d");f.Clock.Advance(TimeSpan.FromSeconds(1));f.Text("💻");f.Undo();Check(f.Value=="👩\u200d","undo cross-transaction ZWJ suffix");f.Undo();Check(f.Value=="👩","undo ZWJ joiner");f.Undo();Check(f.Value=="","undo original emoji");});
Run("regional",f=>{f.Key(UiKey.Home);f.Text("🇽");Check(f.Value=="🇽🇦🇧","regional-indicator insertion accepted");f.Undo();Check(f.Value=="🇦🇧","undo regional-indicator regrouping in both documents");f.Redo();Check(f.Value=="🇽🇦🇧","redo regional-indicator regrouping");},"🇦🇧");
Run("delete",f=>{f.Key(UiKey.Backspace);f.Clock.Advance(TimeSpan.FromMilliseconds(250));f.Key(UiKey.Backspace);Check(f.Value=="e\u0301","backspace removes whole emoji graphemes");f.Undo();Check(f.Value=="e\u0301😀👩\u200d💻","merged backspace restores exact text");},"e\u0301😀👩\u200d💻");
Run("IME",f=>{var c=f.Client;int e=c.Session.SessionId;c.ApplyComposition([CompositionEvent.Start(e,0,c.Document.Version,0,new(0,2),"nihao")]);c.ApplyComposition([CompositionEvent.Result(e,0,c.Document.Version,1,"你好",nextPreedit:"shi")]);c.ApplyComposition([CompositionEvent.Result(e,0,c.Document.Version,2,"世")]);Check(f.Value=="你好世","segmented IME commit");f.Undo();Check(f.Value=="你好","one segmented IME result per undo");f.Undo();Check(f.Value=="ab","undo prior IME result restores original selection text");f.Redo();f.Redo();Check(f.Value=="你好世","redo segmented IME results");},"ab");
Run("imeKey",f=>{f.Text("a");var c=f.Client;var e=c.Session.SessionId;c.ApplyComposition([CompositionEvent.Start(e,0,c.Document.Version,0,new(1,1),"n")]);var token=new object();f.Runtime.NotifyCompositionKeyOrigin(token,true);f.Key(UiKey.Z,Field.Command,token);Check(f.Value=="a"&&f.Client.IsComposing,"IME consumed modified undo token wins");f.Undo();Check(f.Value==""&&!f.Client.IsComposing,"unowned undo cancels preedit then undoes");});
Run("wrongModifiers",f=>{f.Text("a");f.Key(UiKey.Z,KeyModifiers.Control|KeyModifiers.Alt);Check(f.Value=="a","AltGr does not trigger undo");f.Key(UiKey.Z,OperatingSystem.IsMacOS()?KeyModifiers.Control:KeyModifiers.Meta);Check(f.Value=="a","wrong platform modifier does not trigger undo");f.Key(UiKey.Z,Field.Command|KeyModifiers.CapsLock);Check(f.Value=="","CapsLock does not prevent undo");});
Run("committedOnly",f=>{f.Text("a");f.Runtime.NotifyWindowActivation(false);f.Runtime.NotifyWindowActivation(true);f.Text("b");f.Undo();Check(f.Value=="a","window activation breaks merge without composition service");f.Undo();Check(f.Value=="","committed-only fallback supports undo");},input:false);
Run("countBudget",f=>{for(int i=0;i<105;i++){f.Clipboard.SetText("x");f.Key(UiKey.V,Field.Command);}for(int i=0;i<100;i++)f.Undo();Check(f.Value=="xxxxx","100-entry budget evicts oldest complete transactions");var n=f.Box.Writes;f.Undo();Check(f.Value=="xxxxx"&&f.Box.Writes==n,"evicted transaction cannot be undone");});
Console.WriteLine($"Checks: {passed} passed, {failed} failed");Environment.ExitCode=failed==0?0:1;

sealed class Box{public string Value="";public int Writes;public bool Reject,ThrowGet,ThrowAfterWrite;public Func<string,string>? Transform;}
sealed class Field:IDisposable{
 public static KeyModifiers Command=>OperatingSystem.IsMacOS()?KeyModifiers.Meta:KeyModifiers.Control;
 public Box Box{get;}=new();public TestUiClock Clock{get;}=new();public StubClipboardService Clipboard{get;}=new();public FakeInput Input{get;}=new();public UiRuntime Runtime{get;}public string Value=>Box.Value;public ITextInputClient Client=>Input.ActiveClient!;
 public Field(string initial,int max,bool input=true){Box.Value=initial;var b=new Binding<string>(()=>Box.ThrowGet?throw new InvalidOperationException():Box.Value,v=>{Box.Writes++;if(!Box.Reject)Box.Value=Box.Transform?.Invoke(v)??v;if(Box.ThrowAfterWrite)throw new InvalidOperationException();});Runtime=new UiRuntime(new TextField(b){MaxLength=max},new UiRuntimeServices{Dispatcher=new HeadlessUiDispatcher(),TextService=new StubTextService(),Clock=Clock,Clipboard=Clipboard,TextInputService=input?Input:null});Runtime.SetViewport(new Size(400,80),1);Runtime.Start();Runtime.ProcessPendingWork();Runtime.BuildFrame();Runtime.SetFocus(Runtime.Root);Key(UiKey.End);}
 public void Text(string s){Runtime.DispatchTextInput(new TextInputEvent(s));Runtime.ProcessPendingWork();}
 public void Key(UiKey k,KeyModifiers m=KeyModifiers.None,object? token=null){Runtime.DispatchKey(new KeyEvent(KeyEventKind.Down,k,m),token);Runtime.DispatchKey(new KeyEvent(KeyEventKind.Up,k,m),token);Runtime.ProcessPendingWork();}
 public void Undo()=>Key(UiKey.Z,Command);public void Redo()=>Key(UiKey.Z,Command|KeyModifiers.Shift);public void Dispose(){Box.ThrowGet=false;Box.ThrowAfterWrite=false;Runtime.Dispose();}
}
sealed class FakeInput:ITextInputService{public TextInputCapabilities Capabilities=>TextInputCapabilities.Full;public ITextInputClient? ActiveClient{get;private set;}public bool StartSession(ITextInputClient c){ActiveClient=c;return true;}public bool EndSession(){ActiveClient=null;return true;}public TextInputApplyResult ApplyComposition(IReadOnlyList<CompositionEvent> e)=>ActiveClient!.ApplyComposition(e);public bool RequestCancel()=>ActiveClient?.CancelComposition()??false;public bool PublishGeometry(TextInputGeometry g)=>false;}
